def fibonacci_sequence(start_sequence: list[int], length: int) -> list[int]:
    """斐波拉契数列 从第三项开始，每一项为前两项之和
    
    Args:
        start_sequence (liist[int]): 斐波拉契数列第一项和第二项的列表
        length (int): 这个数列的长度

    Returns:
        list[int]: 指定长度的列表，包括传入的start_sequence
    """
    fib_list = start_sequence[:]
    if length == 0:
        return []
    elif length <= 2:
        return fib_list[:length]
    else:
        for i in range(2, length):
            fib_list.append(fib_list[i - 1] + fib_list[i - 2])
        return fib_list
    
if __name__ == '__main__':
    print(fibonacci_sequence([0, 1], 20))
    print(fibonacci_sequence([21, 32], 1))
