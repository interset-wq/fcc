def jbelmu(text: str) -> str:
    parts = text.split()
    for i in range(len(parts)):
        part = parts[i]
        if len(part) <= 3:
            continue
        chars = list(part)
        head = chars.pop(0)
        tail = chars.pop()
        parts[i] = head + ''.join(sorted(chars)) + tail
    text = ' '.join(parts)
    return text
