function areAnagrams(str1, str2) {
    // 辅助函数：处理字符串（转为小写、去除空格、排序字符）
    function processString(str) {
        // 转为小写，去除所有空格，然后拆分为字符数组
        return str.toLowerCase().replace(/\s+/g, '').split('')
            // 排序字符数组
            .sort()
            // 重新组合为字符串
            .join('');
    }
    
    // 处理两个输入字符串并比较结果
    return processString(str1) === processString(str2);
}



console.log(areAnagrams("listen", "silent")); // true
console.log(areAnagrams("School master", "The classroom"))
